Interface PendexTheme

The full schema a src/themes/*.toml file may satisfy: the ten required ThemeColors slots plus optional metadata and extended color tables. Extended tables aren't consumed by buildTheme — they're captured for future use (e.g. via ThemeManager.extended()) without forcing every theme author to define slots nothing reads yet.

interface PendexTheme {
    name: string;
    author?: string;
    version?: string;
    description?: string;
    colors: ThemeColors;
    palette?: Record<string, string>;
    website?: Record<string, string>;
    terminal?: Record<string, string>;
    ansi?: Record<string, string>;
    syntax?: Record<string, string>;
    git?: Record<string, string>;
    diagnostics?: Record<string, string>;
    diff?: Record<string, string>;
    brand?: Record<string, string>;
}
