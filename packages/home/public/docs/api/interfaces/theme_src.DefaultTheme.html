Interface DefaultTheme

The fully-resolved, ready-to-use theme: every slot is a function that takes plain text and returns ANSI-styled text. This is what buildTheme produces and what the rest of the app (via useTheme) actually calls.

interface DefaultTheme {
    primary(text: string): string;
    secondary(text: string): string;
    success(text: string): string;
    warning(text: string): string;
    error(text: string): string;
    info(text: string): string;
    muted(text: string): string;
    title(text: string): string;

    bold(text: string): string;
    italic(text: string): string;
    textDecoration(text: string): string;
    textTransform(text: string): string;
}
